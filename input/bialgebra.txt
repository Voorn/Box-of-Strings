# Bialgebra example #

import theory/symmetry


op c : 1 -> 2 [2]
op d : 1 -> 0 [2]
op e : 2 -> 1 [w]
op b : 0 -> 1 [w]

schema naturality {x,c} {}
schema naturality {x,d} {}
schema naturality {x,e} {}
schema naturality {x,b} {}
schema mirror {x,c,d,e,b} {x,e,b,c,d}

# associative #
def associative (3) {0:e,0:e = 1:e,0:e}

# unital # 
def unital (1) {0:b,0:e = id}

# commutative #
def commutative (2) {0:x,0:e = 0:e}

lem  doub-bial (3) {
    0:e, 0:e, 0:c, 0:c
    =
    0:c, 0:c, 3:c, 3:c, 6:c, 6:c, 2:x, 1:x, 3:x, 0:e, 4:x, 3:x, 2:x, 1:x, 2:e, 4:x, 3:x, 4:e, 0:e, 1:e, 2:e
}

	=
	0:[Dist(x,e,F(I)) (2) {0:e, 0:<I (1) {0:c, 0:c}> = 0:<I (1) {0:c, 0:c}>, 3:<I (1) {0:c, 0:c}>, 2:x, 1:x, 3:x, 0:e, 1:e, 2:e}], 3:<I (1) {0:c, 0:c}>, 2:x, 1:x, 3:x, 0:e, 1:e, 2:e
	=
	0:[Id-Func (1) {0:<I (1) {0:c, 0:c}> = 0:c, 0:c}], 3:<I (1) {0:c, 0:c}>, 2:x, 1:x, 3:x, 0:e, 1:e, 2:e, 3:<I (1) {0:c, 0:c}>, 2:x, 1:x, 3:x, 0:e, 1:e, 2:e
	=
	0:c, 0:c, 3:[Id-Func (1) {0:<I (1) {0:c, 0:c}> = 0:c, 0:c}], 2:x, 1:x, 3:x, 0:e, 1:e, 2:e, 3:<I (1) {0:c, 0:c}>, 2:x, 1:x, 3:x, 0:e, 1:e, 2:e
	=
	0:c, 0:c, 3:c, 3:c, 6:[Id-Func (1) {0:<I (1) {0:c, 0:c}> = 0:c, 0:c}], 2:x, 1:x, 3:x, 0:e, 1:e, 2:e, 2:x, 1:x, 3:x, 0:e, 1:e, 2:e

	=
	0:c, 0:c, 3:c, 3:c, 6:c, 6:c, 2:x, 1:x, 3:x, 0:e, 3:[SymD(x,e) (3) {0:e, 0:x = 1:x, 0:x, 1:e}], 2:x, 1:x, 2:e, 3:x, 0:e, 1:e, 2:e
	=

}
