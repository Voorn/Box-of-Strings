# Bigger and better circuits  #

# Import the copy-discard category theory #
import  circuit/base-gates


# XOR (composite) #
op  r : 2 -> 1  [xor]


def  XOR-definition (2) {
        0:r
        =
        0:c, 2:c, 1:a, 1:n, 1:c, 0:a, 1:a, 0:o
}

lem  XOR-commutative (2) {
        0:x, 0:r
        =
        0:r
}

lem  XOR-alt (2) {
        0:r
        =
        0:c, 2:c, 1:x, 2:a, 0:o, 1:n, 0:a
}


schema  flip  {r} {r}

lem  XOR-char2 (1) {
        0:c, 0:r
        =
        0:d, 0:0
}


lem  TRUE-XOR (1) {
        0:1, 0:r
        =
        0:n
}

lem  XOR-AND-dist (3) {
        2:c, 1:x, 0:a, 1:a, 0:r
        =
        0:r, 0:a
}


lem  XOR-associative (3) {
        0:r, 0:r
        =
        1:r, 0:r
}

op  f : 3 -> 2

note {We define a full adder accepting three bits as input,|computing their sum in terms of a two digit binary number.}
def  Full adder (3) {
        0:f
        =
        0:c, 2:c, 4:c, 1:x, 3:x, 2:x, 0:r, 0:r, 1:c, 3:c, 5:c, 2:x, 4:x, 1:a, 2:a, 3:a, 1:o, 1:o
}
