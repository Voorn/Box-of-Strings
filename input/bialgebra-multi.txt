# Multi Bialgebra #

import bialgebra

op  C : 1 -> 3 [2]

op  E : 3 -> 1 [w]

op  X : 3 -> 3 [3]

def  MultiCopy (1) {
    0:C
    =
    0:c, 0:c
}

def  MultiSum (3) {
    0:E
    =
    0:e, 0:e
}

def  MultiSwap (3) {
    0:X
    =
    0:x, 1:x, 0:x
}

lem  MultiBialgebra (3) {
    0:E, 0:C
    =
    0:C, 3:C, 6:C, 2:x, 5:x, 1:x, 3:X, 6:x, 2:x, 5:x, 0:E, 1:E, 2:E
}
